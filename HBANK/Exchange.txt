    REASON_FOR_REJECTION STRING OPTIONS(description="Reason if claim was rejected (optional)"),

    SOURCE_OF_THE_CLAIM STRING OPTIONS(description="Source code indicating where claim originated from"),

    TOTAL_OFFERED_AMOUNT STRING OPTIONS(description="Total amount offered in settlement"),

    TOTAL_WRITTEN_OFF STRING OPTIONS(description="Amount written off/written out of accounts"),

    TRANSACTION_VALUE_ON_CARD STRING OPTIONS(description="Value of disputed transaction recorded on card"),

    WRITE_OFF_CATEGORY STRING OPTIONS(description="Category of write-off (optional)"),

    WRITE_OFF_NUMBER STRING OPTIONS(description="Reference number for write-off transaction (optional)"),

    STATUS_CODE STRING OPTIONS(description="Status of the record (Active, Inactive, etc.)"),

    MODIFIED_ON STRING OPTIONS(description="Timestamp of last modification"),

    MODIFIED_BY STRING OPTIONS(description="User ID or name of person who last modified the record (optional)"),

    GCP_LOAD_DATETIME STRING OPTIONS(description="Timestamp when data was loaded into GCP (Google Cloud Platform)"),

    PRM_LNE_OF_BUS_CDE_FILTER STRING OPTIONS(description="The line of business of the customer - e.g. RBWM, CMB, GPB, GB&M"),

    ENTITY_CDE_FILTER STRING OPTIONS(description="Entity code as defined within ESR based on the value of institution ID"),

    CTRY_CDE_FILTER STRING OPTIONS(description="The country the customer is domiciled in, used for downstream filtering"),

    LAST_INGESTION_TIMESTAMP TIMESTAMP OPTIONS(description="Timestamp when record was last inserted or updated within Big Query itself")
