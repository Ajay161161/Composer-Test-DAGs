CREATE TABLE `${STAGING_DATASET}.CLAIMS_HISTORY`
(

    PRODUCT_NAME STRING OPTIONS(description="Name of the financial product tied to the case (e.g., credit card type)."),

    RETAILER STRING OPTIONS(description="Merchant/retailer involved in the disputed transaction."),

    TOTAL_TRANSACTION_VALUE STRING OPTIONS(description="Total value of all transactions under dispute for the case."),

    TRANSACTION_VALUE_ON_CARD STRING OPTIONS(description="Portion of disputed value paid by card."),

    UK_CARD STRING OPTIONS(description="Amount/count of UK transactions paid by card."),

    UK_NON_CARD STRING OPTIONS(description="Amount/count of UK transactions not paid by card (e.g., bank transfer, cash)."),

    FOREIGN_CARD STRING OPTIONS(description="Amount/count of non-UK transactions paid by card."),

    FOREIGN_NON_CARD STRING OPTIONS(description="Amount/count of non-UK transactions not paid by card."),

    AMOUNT_CHARGEDBACK_ALREADY STRING OPTIONS(description="Amount already recovered or processed through chargeback."),

    DATE_CASE_COMPLETE STRING OPTIONS(description="Date the case was fully completed/closed operationally."),

    DAY_CASE_COMPLETED STRING OPTIONS(description="Day of week the case was completed."),

    AMOUNT_SAVED_CCA_NEGOTIATIONS STRING OPTIONS(description="Amount saved through CCA-related negotiations."),

    AMOUNT_SAVED_CBKS STRING OPTIONS(description="Amount saved through chargebacks."),

    AMOUNT_SAVED_PRE_NEG STRING OPTIONS(description="Amount saved through pre-negotiation actions."),

    AMOUNT_SAVED_RES_WITH_RETAILER STRING OPTIONS(description="Amount saved through direct resolution with retailer."),

    AMOUNT_SAVED_A_COURT STRING OPTIONS(description="Amount saved/awarded through court action."),

    BUSINESS_ERROR_OVER_1K STRING OPTIONS(description="Flag indicating whether business error impact exceeded 1,000 (currency units)."),

    BUSINESS_AREA_AT_FAULT STRING OPTIONS(description="Internal business unit identified as responsible for error/failure."),

    TOTAL_WRITTEN_OFF STRING OPTIONS(description="Total amount written off on the case."),

    WRITE_OFF_CATEGORY STRING OPTIONS(description="Classification/reason code for write-off."),

    FOS_CASE STRING OPTIONS(description="Flag indicating whether escalated to Financial Ombudsman Service."),

    COURT_CASE STRING OPTIONS(description="Flag indicating whether legal/court proceedings were initiated."),

    NOTES STRING OPTIONS(description="Free-text case notes and handling commentary."),

    FOREIGN_TRANSACTION STRING OPTIONS(description="Flag indicating case involves at least one foreign transaction."),

    MERCHANT_TRADING STRING OPTIONS(description="Merchant trading status/context (e.g., active, ceased trading, insolvent)."),

    DATE_OF_OFFER STRING OPTIONS(description="Date a settlement/compensation offer was made."),

    TOTAL_OFFERED_AMOUNT STRING OPTIONS(description="Total amount offered to customer for settlement/redress."),

    REASON_FOR_CLAIM STRING OPTIONS(description="Primary reason the customer raised the claim/dispute."),

    CONSEQUENTIAL_LOSS STRING OPTIONS(description="Claimed consequential loss amount/details tied to the incident."),

    SOURCE_OF_CLAIM STRING OPTIONS(description="Channel/source where claim originated (e.g., branch, phone, online, FOS)."),

    CUST_NAME STRING OPTIONS(description="Customer full name."),

    AMNT_REC_CHGB STRING OPTIONS(description="Amount recovered via chargeback."),

    AMNT_PAID_CD STRING OPTIONS(description="Amount paid under card dispute/chargeback decision (or card dispute payout)."),

    CONSEQ_LOSS STRING OPTIONS(description="Short form of consequential loss field (amount/value)."),

    CASE_CREATED_BY STRING OPTIONS(description="User/agent ID who created the case record."),

    CASE_CLOSED_BY STRING OPTIONS(description="User/agent ID who closed the case."),

    CLOSE_REF STRING OPTIONS(description="Closure reference number/code."),

    STATUS STRING OPTIONS(description="Current lifecycle status of case (e.g., Open, In Progress, Closed)."),

    FILE_DATE STRING OPTIONS(description="Date case file/document was created or formally filed."),

    PRM_LNE_OF_BUS_CDE_FILTER STRING OPTIONS(description="The line of business of the customer - e.g. RBWM, CMB, GPB, GB&M"),

    ENTITY_CDE_FILTER STRING OPTIONS(description="Entity code as defined within ESR based on the value of institution ID"),

    CTRY_CDE_FILTER STRING OPTIONS(description="The country the customer is domiciled in, used for downstream filtering"),

    LAST_INGESTION_TIMESTAMP TIMESTAMP OPTIONS(description="Timestamp when record was last inserted or updated within Big Query itself")
);
