DECLARE TBL_NAME STRING DEFAULT 'CLAIMS';

DECLARE DATASETNAME STRING DEFAULT '${TARGET_DATASET}';

DECLARE DATEFORMAT STRING DEFAULT '%d-%b-%Y';

INSERT INTO ${STAGING_DATASET}.CLAIMS
SELECT *
FROM ${STAGING_DATASET}.CLAIMS_UNTAGGED;

DELETE FROM ${STAGING_DATASET}.CLAIMS_UNTAGGED
WHERE TRUE;

UPDATE ${STAGING_DATASET}.CLAIMS CL

SET

CL.PRM_LNE_OF_BUS_CDE_FILTER =
CASE
    WHEN UPPER(TRIM(CL.ENTITY)) = 'CMB' THEN 'CMB'
    ELSE 'RBWM'
END,

CL.ENTITY_CDE_FILTER = 'HBEU',

CL.CTRY_CDE_FILTER = 'UK',

CL.LAST_INGESTION_TIMESTAMP = CURRENT_TIMESTAMP()

WHERE CL.PRM_LNE_OF_BUS_CDE_FILTER IS NULL;

INSERT INTO ${TARGET_DATASET}.CLAIMS
(
    INSTANCE_ID,
    DATE_CASE_RECEIVED,
    BRAND,
    CIN,
    ENTITY,
    ACCOUNT_NUMBER,
    RETAILER,
    TOTAL_TRANSACTION_VALUE,
    CUSTOMER_NAME,
    CLAIM_STATUS,
    AMOUNT_CLAIMED,
    AMOUNT_SAVED_CHBK,
    AMOUNT_SAVED_RE_VIA_COURT,
    AMOUNT_SAVED_RES_WITH_RETAILER,
    ARCHIVED,
    BUSINESS_AREA_AT_FAULT,
    CASE_COMPLETE_BY,
    CATEGORY_NAME,
    COMPLETE,
    CONSEQUENTIAL_LOSS,
    COURT_CASE,
    DATE_CASE_COMPLETE,
    DATE_OF_OFFER,
    FOS_CASE,
    LEGAL_REFERRAL,
    NOTES,
    PRODUCT,
    REASON_FOR_CLAIM,
    REASON_FOR_REJECTION,
    SOURCE_OF_THE_CLAIM,
    TOTAL_OFFERED_AMOUNT,
    TOTAL_WRITTEN_OFF,
    TRANSACTION_VALUE_ON_CARD,
    WRITE_OFF_CATEGORY,
    WRITE_OFF_NUMBER,
    STATUS_CODE,
    MODIFIED_ON,
    MODIFIED_BY,
    GCP_LOAD_DATETIME,
    PRM_LNE_OF_BUS_CDE_FILTER,
    ENTITY_CDE_FILTER,
    CTRY_CDE_FILTER,
    LAST_INGESTION_TIMESTAMP
)

SELECT

    LTRIM(RTRIM(INSTANCE_ID)),

    PARSE_DATE(DATEFORMAT, LTRIM(RTRIM(DATE_CASE_RECEIVED))),

    LTRIM(RTRIM(BRAND)),

    LTRIM(RTRIM(CIN)),

    LTRIM(RTRIM(ENTITY)),

    LTRIM(RTRIM(ACCOUNT_NUMBER)),

    LTRIM(RTRIM(RETAILER)),

    CAST(REGEXP_REPLACE(TOTAL_TRANSACTION_VALUE, r',', '') AS NUMERIC),

    LTRIM(RTRIM(CUSTOMER_NAME)),

    LTRIM(RTRIM(CLAIM_STATUS)),

    CAST(REGEXP_REPLACE(AMOUNT_CLAIMED, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(AMOUNT_SAVED_CHBK, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(AMOUNT_SAVED_RE_VIA_COURT, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(AMOUNT_SAVED_RES_WITH_RETAILER, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(ARCHIVED, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(BUSINESS_AREA_AT_FAULT, r',', '') AS NUMERIC),

    LTRIM(RTRIM(CASE_COMPLETE_BY)),

    LTRIM(RTRIM(CATEGORY_NAME)),

    CAST(REGEXP_REPLACE(COMPLETE, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(CONSEQUENTIAL_LOSS, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(COURT_CASE, r',', '') AS NUMERIC),

    PARSE_DATE(DATEFORMAT, LTRIM(RTRIM(DATE_CASE_COMPLETE))),

    PARSE_DATE(DATEFORMAT, LTRIM(RTRIM(DATE_OF_OFFER))),

    CAST(REGEXP_REPLACE(FOS_CASE, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(LEGAL_REFERRAL, r',', '') AS NUMERIC),

    LTRIM(RTRIM(NOTES)),

    LTRIM(RTRIM(PRODUCT)),

    LTRIM(RTRIM(REASON_FOR_CLAIM)),

    LTRIM(RTRIM(REASON_FOR_REJECTION)),

    CAST(REGEXP_REPLACE(SOURCE_OF_THE_CLAIM, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(TOTAL_OFFERED_AMOUNT, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(TOTAL_WRITTEN_OFF, r',', '') AS NUMERIC),

    CAST(REGEXP_REPLACE(TRANSACTION_VALUE_ON_CARD, r',', '') AS NUMERIC),

    LTRIM(RTRIM(WRITE_OFF_CATEGORY)),

    LTRIM(RTRIM(WRITE_OFF_NUMBER)),

    LTRIM(RTRIM(STATUS_CODE)),

    PARSE_DATE(DATEFORMAT, LTRIM(RTRIM(MODIFIED_ON))),

    LTRIM(RTRIM(MODIFIED_BY)),

    PARSE_DATE(DATEFORMAT, LTRIM(RTRIM(GCP_LOAD_DATETIME))),

    PRM_LNE_OF_BUS_CDE_FILTER,

    ENTITY_CDE_FILTER,

    CTRY_CDE_FILTER,

    LAST_INGESTION_TIMESTAMP

FROM ${STAGING_DATASET}.CLAIMS

WHERE PRM_LNE_OF_BUS_CDE_FILTER IS NOT NULL
AND TRIM(PRM_LNE_OF_BUS_CDE_FILTER) <> 'NA'
AND TRIM(CTRY_CDE_FILTER) <> 'FR';

INSERT INTO ${STAGING_DATASET}.CLAIMS_UNTAGGED

SELECT *
FROM ${STAGING_DATASET}.CLAIMS

WHERE PRM_LNE_OF_BUS_CDE_FILTER IS NULL;

DELETE FROM ${STAGING_DATASET}.CLAIMS

WHERE PRM_LNE_OF_BUS_CDE_FILTER IS NOT NULL;
