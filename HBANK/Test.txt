def decode_base64_gcs_file(bucket_name, source_gcs_path, decoded_folder):

    global DECODED_FILE_CREATED

    try:

        if 'liveproving' in bucket_name:
            sa_key = get_sa_key()
            storage_client = storage.Client.from_service_account_info(sa_key)
        else:
            storage_client = storage.Client()

        bucket = storage_client.bucket(bucket_name)

        source_blob = bucket.blob(source_gcs_path)

        if not source_blob.exists():
            print(f"Source encoded file not found in GCS: {source_gcs_path}")
            return -1

        print(f"Reading Base64 encoded file from: gs://{bucket_name}/{source_gcs_path}")

        encoded_bytes = source_blob.download_as_bytes()

        encoded_bytes = encoded_bytes.strip()

        try:
            decoded_bytes = base64.b64decode(encoded_bytes, validate=True)
            print("Base64 standard decode successful.")
        except Exception as decode_err:
            print(f"Standard Base64 decode failed: {decode_err}. Attempting URL-safe Base64...")
            try:
                decoded_bytes = base64.b64decode(encoded_bytes, altchars=b'-_')
                print("URL-safe Base64 decode successful.")
            except Exception as fallback_err:
                print(f"URL-safe Base64 decode also failed: {fallback_err}")
                return -1

        file_name_only = os.path.basename(source_gcs_path)

        decoded_blob_path = f"{decoded_folder}/{file_name_only}"

        print(f"Writing decoded file to: gs://{bucket_name}/{decoded_blob_path}")

        destination_blob = bucket.blob(decoded_blob_path)

        destination_blob.upload_from_string(decoded_bytes, content_type='text/csv')

        DECODED_FILE_CREATED = True

        print(f"Base64 decode completed. Decoded file at: gs://{bucket_name}/{decoded_blob_path}")

        return decoded_blob_path

    except Exception as e:
        print(f"Unexpected error in decode_base64_gcs_file: {e}")
        return -1
